if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  message(FATAL_ERROR "Configure from the RUPTURA project root: cmake -S ../.. -B build. The shared model reuses the original main material laws.")
endif()
set(RUPTURA_PROCESS_ROOT "${PROJECT_SOURCE_DIR}")
find_package(ZLIB REQUIRED)
file(GLOB yaml_sources "${RUPTURA_PROCESS_ROOT}/third_party/libyaml/src/*.c")
add_library(ruptura_yaml STATIC ${yaml_sources})
target_include_directories(ruptura_yaml PUBLIC "${RUPTURA_PROCESS_ROOT}/third_party/libyaml/include" PRIVATE "${RUPTURA_PROCESS_ROOT}/third_party/libyaml/src")
target_compile_definitions(ruptura_yaml PRIVATE HAVE_CONFIG_H=0 _POSIX_C_SOURCE=200809L YAML_VERSION_STRING="0.2.5" YAML_VERSION_MAJOR=0 YAML_VERSION_MINOR=2 YAML_VERSION_PATCH=5)
file(GLOB sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
add_library(ruptura_process STATIC ${sources})
set_target_properties(ruptura_process ruptura_yaml PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(ruptura_process PUBLIC "${RUPTURA_PROCESS_ROOT}/include" "${RUPTURA_PROCESS_ROOT}/third_party")
target_link_libraries(ruptura_process PUBLIC ruptura_yaml ZLIB::ZLIB ${CMAKE_DL_LIBS})
if(NOT TARGET SUNDIALS::cvode AND (CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR OR BUILD_SUNDIALS))
  find_package(SUNDIALS 7.5 CONFIG QUIET)
endif()
set(process_cvode OFF)
if(TARGET SUNDIALS::cvode AND TARGET SUNDIALS::sunmatrixdense AND TARGET SUNDIALS::sunlinsoldense)
  set(process_cvode ON)
  target_link_libraries(ruptura_process PUBLIC SUNDIALS::cvode SUNDIALS::nvecserial SUNDIALS::sunmatrixdense SUNDIALS::sunlinsoldense SUNDIALS::sunlinsolspgmr)
  if(TARGET SUNDIALS::core)
    target_link_libraries(ruptura_process PUBLIC SUNDIALS::core)
  endif()
endif()
target_compile_definitions(ruptura_process PUBLIC RUPTURA_HAS_SUNDIALS=$<BOOL:${process_cvode}>)
if(BUILD_TESTING)
  foreach(name test_shared_chemisorption test_mpd test_shared_column test_shared_numerics test_kinetics_tea test_literature_boundaries test_upstream_material_laws test_main_fvm test_flow_controls test_equilibrium test_column test_integrator test_diagnostics test_jacket test_integrator_cvode test_npz test_runner)
    add_executable(process_${name} "${RUPTURA_PROCESS_ROOT}/tests/process/${name}.cpp")
    target_link_libraries(process_${name} PRIVATE ruptura_process)
    if(name STREQUAL "test_kinetics_tea")
      add_test(NAME process_${name} COMMAND process_${name} "${RUPTURA_PROCESS_ROOT}/tests/process/kinetics_tea_reference.json")
    elseif(name STREQUAL "test_equilibrium")
      add_test(NAME process_${name} COMMAND process_${name} "${RUPTURA_PROCESS_ROOT}/tests/process/equilibrium_reference.json")
    elseif(name STREQUAL "test_column")
      add_test(NAME process_${name} COMMAND process_${name} "${RUPTURA_PROCESS_ROOT}/tests/process/column_reference.json")
    elseif(name STREQUAL "test_npz")
      add_test(NAME process_${name} COMMAND process_${name} "${CMAKE_BINARY_DIR}/process_npz_test")
    else()
      add_test(NAME process_${name} COMMAND process_${name})
    endif()
    set_tests_properties(process_${name} PROPERTIES WORKING_DIRECTORY "${RUPTURA_PROCESS_ROOT}")
  endforeach()
endif()
